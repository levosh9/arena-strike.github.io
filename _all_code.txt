Game.syncAmmoToSlot = function(){
  const p = Game.player;
  const s = p.slots[p.currentSlot];
  if (s){ s.ammo = p.ammo; s.reserve = p.reserve; }
};

Game.activateSlot = function(slotKey){
  const p = Game.player;
  const s = p.slots[slotKey];
  if (!s) return;
  p.currentSlot = slotKey;
  p.prevSlot = slotKey;
  p.weapon = s.key;
  p.ammo = s.ammo;
  p.reserve = s.reserve;
  p.reloading = false;
  if (Game.equipGun) Game.equipGun(s.key);
};

Game.switchSlot = function(slotKey){
  const p = Game.player;
  if (slotKey === p.currentSlot || !p.slots[slotKey]) return;
  if (p.ads) Game.exitADS();
  Game.syncAmmoToSlot();
  Game.activateSlot(slotKey);
  if (Game.updHUD) Game.updHUD();
};

// Переключение и циклический выбор гранат на клавишу [4]
Game.cycleGrenadeSlot = function(){
  const p = Game.player;
  const owned = Game.getOwnedGrenades(p);
  if (!owned.length){
    if (Game.showMsg) Game.showMsg('Нет гранат — купите в магазине [B]', 1400);
    return;
  }

  if (p.ads) Game.exitADS();

  if (p.currentSlot !== 'grenade'){
    Game.syncAmmoToSlot();
    p.prevSlot = p.currentSlot;
    p.currentSlot = 'grenade';
    p.reloading = false;
    p.activeGrenade = owned.includes(p.activeGrenade) ? p.activeGrenade : owned[0];
  } else {
    const idx = owned.indexOf(p.activeGrenade);
    p.activeGrenade = owned[(idx + 1) % owned.length];
  }

  if (Game.equipGun) Game.equipGun('grenade_' + p.activeGrenade);
  if (Game.updHUD) Game.updHUD();
};

Game.refreshCrosshair = function(){
  const ch = Game.$('crosshair');
  const p = Game.player;
  if (p.dead || p.isScoped || (p.currentSlot !== 'grenade' && Game.isSniper(p.weapon))){
    ch.style.display = 'none';
  } else {
    ch.style.display = '';
  }
};

Game.enterADS = function(){
  const p = Game.player;
  if (p.dead || p.reloading || Game.state.phase !== 'live' || p.currentSlot === 'grenade' || Game.isKnife(p.weapon)) return;
  const scoped = Game.isSniper(p.weapon);
  p.ads = true;
  p.isScoped = scoped;
  if (scoped){
    Game.camera.fov = 30;
    Game.$('scope').classList.add('active');
  } else {
    Game.camera.fov = 60;
  }
  Game.camera.updateProjectionMatrix();
  Game.refreshCrosshair();
};

Game.tone = function(freq, dur, type = 'square', vol = 0.08){
  if (!AC || !Game.settings.soundOn || vol <= 0.002) return;
  vol *= Game.settings.volume;
  if (vol <= 0.002) return;
  const o = AC.createOscillator(), g = AC.createGain();
  o.type = type;
  o.frequency.value = freq;
  g.gain.setValueAtTime(vol, AC.currentTime);
  g.gain.exponentialRampToValueAtTime(0.001, AC.currentTime + dur);
  o.connect(g).connect(AC.destination);
  o.start();
  o.stop(AC.currentTime + dur);
};

  function addRampMesh(r){
    const w = r.maxX - r.minX;
    const d = r.maxZ - r.minZ;
    const steps = 8;
    const mat = new THREE.MeshLambertMaterial({ color: r.color });
    for (let i = 0; i < steps; i++){
      const t = (i + 0.5) / steps;
      const curH = r.h0 + (r.h1 - r.h0) * t;
      if (curH <= 0.02) continue;
      if (r.axis === 'z'){
        const segD = d / steps;
        const cz = r.minZ + (r.dir < 0 ? (1 - t) * d : t * d);
        const m = new THREE.Mesh(new THREE.BoxGeometry(w, curH, segD), mat);
        m.position.set((r.minX + r.maxX) * 0.5, curH * 0.5, cz);
        mapGroup.add(m);
        obstacleMeshes.push(m);
      } else {
        const segW = w / steps;
        const cx = r.minX + (r.dir < 0 ? (1 - t) * w : t * w);
        const m = new THREE.Mesh(new THREE.BoxGeometry(segW, curH, d), mat);
        m.position.set(cx, curH * 0.5, (r.minZ + r.maxZ) * 0.5);
        mapGroup.add(m);

      }
    }

  Game.getGroundHeight = function(x, z){
    let h = 0;
    for (const p of activePlatforms){
      if (x >= p.minX && x <= p.maxX && z >= p.minZ && z <= p.maxZ){
        if (p.h > h) h = p.h;
      }
    }
    for (const r of activeRamps){
      if (x >= r.minX && x <= r.maxX && z >= r.minZ && z <= r.maxZ){
        let t = 0;
        if (r.axis === 'z'){
          t = r.dir < 0 ? (r.maxZ - z) / (r.maxZ - r.minZ) : (z - r.minZ) / (r.maxZ - r.minZ);
        } else {
          t = r.dir < 0 ? (r.maxX - x) / (r.maxX - r.minX) : (x - r.minX) / (r.maxX - r.minX);
        }
        t = Math.max(0, Math.min(1, t));
        const rh = r.h0 + (r.h1 - r.h0) * t;
        if (rh > h) h = rh;
      }
    }
    return h;

  Game.isBlockedBySmoke = function(ax, az, bx, bz){
    if (!Game.smokeClouds || !Game.smokeClouds.length) return false;
    const dx = bx - ax, dz = bz - az;
    const lenSq = dx * dx + dz * dz;
    for (const sm of Game.smokeClouds){
      const r = sm.radius;
      let t = 0;
      if (lenSq > 0.0001){
        t = Math.max(0, Math.min(1, ((sm.pos.x - ax) * dx + (sm.pos.z - az) * dz) / lenSq));
      }
      const cx = ax + t * dx;
      const cz = az + t * dz;
      if (Math.hypot(sm.pos.x - cx, sm.pos.z - cz) < r) return true;
    }
    return false;
  };

    }

    const extraAfterBuy = Math.max(0, CFG.BUY_WINDOW - CFG.BUY_TIME);
    const timeLeft = st.phase === 'buy'
      ? Math.ceil(st.phaseTimer + extraAfterBuy)
      : Math.ceil(extraAfterBuy - (CFG.ROUND_TIME - st.roundTimeLeft));

      return { ok: false, reason: 'Время закупки истекло (' + CFG.BUY_WINDOW + ' с)', timeLeft: 0 };
    }

    const zoneX = Game.getTeamSpawnX(p.team);

    if (dist > CFG.BUY_ZONE_RADIUS){

  function spawnHitEffect(point, color){
    const mat = new THREE.SpriteMaterial({
      map: Game.flashTexture, color, blending: THREE.AdditiveBlending,
      transparent: true, depthWrite: false, opacity: 0.95
    });
    const s = new THREE.Sprite(mat);
    s.scale.set(0.25, 0.25, 1);
    s.position.copy(point);
    Game.scene.add(s);
    hitEffects.push({ s, life: 0.22, max: 0.22 });
  }

  Game.updateHitEffects = function(dt){
    for (let i = hitEffects.length - 1; i >= 0; i--){
      const e = hitEffects[i];
      e.life -= dt;
      const k = Math.max(0, e.life / e.max);
      e.s.material.opacity = k;
      const sc = 0.25 + (1 - k) * 0.55;
      e.s.scale.set(sc, sc, 1);
      if (e.life <= 0){
        Game.scene.remove(e.s);
        e.s.material.dispose();
        hitEffects.splice(i, 1);
      }
    }
  };

  Game.applyDamage = function(victim, dmg, attacker, head, hitPoint, weaponOverride){
    const ph = Game.state.phase;
    if (victim.dead || ph === 'roundEnd' || ph === 'over' || ph === 'buy') return;
    if (head) dmg *= 4;

    if (victim.armor > 0 && weaponOverride !== 'molotov'){
      const absorbRate = head ? (victim.helmet ? 0.3 : 0) : 0.5;
      if (absorbRate > 0){
        const absorbed = Math.min(victim.armor, dmg * absorbRate);
        victim.armor -= absorbed;
        dmg -= absorbed;
      }
      if (victim.armor <= 0){
        victim.armor = 0;
        victim.helmet = false;
        if (!victim.isPlayer && victim.helmetMesh) victim.helmetMesh.visible = false;
      }
    }

    const realDmg = Math.min(victim.hp, Math.max(0, dmg));
    recordDamageStats(victim, attacker, realDmg);

    victim.hp -= dmg;
    if (weaponOverride !== 'molotov'){
      victim.slowUntil = performance.now() + 320;
    }
    if (hitPoint) spawnHitEffect(hitPoint, head ? 0xffd25e : 0xff5533);
    if (victim.isPlayer && weaponOverride !== 'molotov'){ Game.redFlash(); Game.updHUD(); }
    if (attacker && attacker.isPlayer && !victim.isPlayer && weaponOverride !== 'molotov'){
      Game.hitMark();
      if (head) Game.snd.head(); else Game.snd.hitMark();
    }
    if (victim.hp <= 0) killEntity(victim, attacker, head, weaponOverride);
  };

    const ph = Game.state.phase;
    if (victim.dead || ph === 'roundEnd' || ph === 'over' || ph === 'buy') return;

    let headAfter = headDmg * 4;
    let bodyAfter = bodyDmg;

    if (victim.armor > 0){
      if (victim.helmet && headAfter > 0){
        const headAbsorb = Math.min(victim.armor, headAfter * 0.3);
        victim.armor -= headAbsorb;
        headAfter -= headAbsorb;
      }

      if (bodyAfter > 0){
        const bodyAbsorb = Math.min(victim.armor, bodyAfter * 0.5);
        victim.armor -= bodyAbsorb;
        bodyAfter -= bodyAbsorb;
      }

      if (victim.armor <= 0){
        victim.armor = 0;
        victim.helmet = false;
        if (!victim.isPlayer && victim.helmetMesh) victim.helmetMesh.visible = false;
      }
    }

    const total = headAfter + bodyAfter;
    const realDmg = Math.min(victim.hp, Math.max(0, total));
    recordDamageStats(victim, Game.player, realDmg);

    victim.hp -= total;
    victim.slowUntil = performance.now() + 320;

    if (hitPoint) spawnHitEffect(hitPoint, headDmg > 0 ? 0xffd25e : 0xff5533);
    if (victim.isPlayer){ Game.redFlash(); Game.updHUD(); }
    if (!victim.isPlayer){
      Game.hitMark();
      if (headDmg > 0) Game.snd.head(); else Game.snd.hitMark();
    }
    if (victim.hp <= 0) killEntity(victim, Game.player, headDmg > 0, 'shotgun');
  }

  /* ---- Гранаты: осколочная, световая, дымовая и Молотов ---- */
  Game.spawnGrenade = function(origin, dir, owner, kind = 'he'){
    const mesh = new THREE.Group();
    const bodyCol = kind === 'flash' ? 0xd8dee9 : (kind === 'smoke' ? 0x6b7280 : (kind === 'molotov' ? 0xd97706 : 0x3a4a2a));
    const body = new THREE.Mesh(
      kind === 'he' ? new THREE.SphereGeometry(0.11, 8, 8) : new THREE.CylinderGeometry(0.06, 0.06, 0.22, 8),
      new THREE.MeshLambertMaterial({ color: bodyCol })
    );
    mesh.add(body);
    const pin = new THREE.Mesh(
      new THREE.CylinderGeometry(0.02, 0.02, 0.05, 6),
      new THREE.MeshLambertMaterial({ color: kind === 'smoke' ? 0x4caf50 : (kind === 'flash' ? 0x3b82f6 : (kind === 'molotov' ? 0xff6b00 : 0xff5252)) })
    );
    pin.position.y = 0.13;
    mesh.add(pin);
    mesh.position.copy(origin);
    Game.scene.add(mesh);

    const vel = dir.clone().normalize().multiplyScalar(17);
    vel.y += 2.5;

    const fuse = kind === 'flash' ? Game.CFG.FLASH_FUSE :
                 (kind === 'smoke' ? Game.CFG.SMOKE_FUSE :
                 (kind === 'molotov' ? Game.CFG.MOLOTOV_FUSE : Game.CFG.GRENADE_FUSE));

    grenades.push({
      kind,
      mesh,
      pos: origin.clone(),
      vel,
      fuse,
      owner,
      spinX: (Math.random() - 0.5) * 12,
      spinZ: (Math.random() - 0.5) * 12
    });
  };

  function spawnExplosion(pos, color = 0xff8822, scale = 12){
    const s = new THREE.Sprite(new THREE.SpriteMaterial({
      map: Game.flashTexture, color, blending: THREE.AdditiveBlending,
      transparent: true, depthWrite: false, opacity: 1
    }));
    s.position.copy(pos);
    s.scale.set(1, 1, 1);
    Game.scene.add(s);
    explosions.push({ s, life: 0.45, max: 0.45, maxScale: scale });
  }

  function spawnSmokeCloud(pos){
    const group = new THREE.Group();
    group.position.set(pos.x, 0, pos.z);
    const r = Game.CFG.SMOKE_RADIUS;
    const mat = new THREE.MeshLambertMaterial({ color: 0xbac2cf, transparent: true, opacity: 0.86 });

    const puffs = [
      [0, 1.5, 0, r * 0.62],
      [-r * 0.35, 1.2, -r * 0.25, r * 0.48],
      [ r * 0.35, 1.2,  r * 0.25, r * 0.48],
      [-r * 0.25, 1.3,  r * 0.35, r * 0.46],
      [ r * 0.25, 1.3, -r * 0.35, r * 0.46],
      [0, 2.3, 0, r * 0.45]
    ];
    for (const [px, py, pz, pr] of puffs){
      const m = new THREE.Mesh(new THREE.SphereGeometry(pr, 10, 8), mat);
      m.position.set(px, py, pz);
      group.add(m);
    }

    Game.scene.add(group);
    Game.smokeClouds.push({
      pos: new THREE.Vector3(pos.x, 0, pos.z),
      radius: r,
      life: Game.CFG.SMOKE_DURATION,
      max: Game.CFG.SMOKE_DURATION,
      group,
      mat
    });
  }

  function spawnFireZone(pos, owner){
    if (Game.smokeClouds){
      for (const sm of Game.smokeClouds){
        if (Math.hypot(sm.pos.x - pos.x, sm.pos.z - pos.z) <= sm.radius + 1.5){
          Game.snd.smokePop(pos);
          return;
        }
      }
    }

    const group = new THREE.Group();
    const gy = Game.getGroundHeight(pos.x, pos.z);
    group.position.set(pos.x, gy + 0.03, pos.z);
    const r = Game.CFG.MOLOTOV_RADIUS;

    const baseMat = new THREE.MeshBasicMaterial({ color: 0xff4500, transparent: true, opacity: 0.55 });
    const disc = new THREE.Mesh(new THREE.CircleGeometry(r, 28), baseMat);
    disc.rotation.x = -Math.PI / 2;
    group.add(disc);

    const flameMat = new THREE.MeshBasicMaterial({ color: 0xffa01e, transparent: true, opacity: 0.78 });
    const flames = [];
    for (let i = 0; i < 10; i++){
      const ang = (i / 10) * Math.PI * 2;
      const dist = (i === 0) ? 0 : (0.3 + Math.random() * 0.65) * r;
      const cone = new THREE.Mesh(new THREE.ConeGeometry(0.42, 1.1, 6), flameMat);
      cone.position.set(Math.cos(ang) * dist, 0.5, Math.sin(ang) * dist);
      group.add(cone);
      flames.push(cone);
    }

    Game.scene.add(group);
    fireZones.push({
      pos: new THREE.Vector3(pos.x, gy, pos.z),
      radius: r,
      life: Game.CFG.MOLOTOV_DURATION,
      max: Game.CFG.MOLOTOV_DURATION,
      owner,
      group,
      baseMat,
      flameMat,
      flames,
      tickTimer: 0.25
    });
  }

  function explodeFlash(g){
    spawnExplosion(g.pos, 0xffffff, 8);
    let playerBlinded = false;

    for (const e of Game.allEntities()){
      if (e.dead) continue;
      const d = Game.dist2D(e.pos, g.pos);
      if (d > Game.CFG.FLASH_RADIUS) continue;
      if (!Game.grenadeLOS(g.pos, e)) continue;

      const distFactor = 1 - d / Game.CFG.FLASH_RADIUS;
      const toFlash = new THREE.Vector2(g.pos.x - e.pos.x, g.pos.z - e.pos.z).normalize();
      const lookDir = e.isPlayer
        ? new THREE.Vector2(-Math.sin(e.yaw), -Math.cos(e.yaw))
        : new THREE.Vector2(Math.sin(e.yaw), Math.cos(e.yaw));
      const dot = toFlash.dot(lookDir);

      const facingMul = dot > 0.2 ? 1.0 : (dot > -0.3 ? 0.55 : 0.25);
      const blindSec = (1.2 + 2.8 * distFactor) * facingMul;

      e.blindUntil = Math.max(e.blindUntil || 0, performance.now() + blindSec * 1000);
      if (!e.isPlayer){
        e.target = null;
        e.slowUntil = Math.max(e.slowUntil || 0, performance.now() + blindSec * 600);
      } else {
        playerBlindAlpha = Math.min(1, Math.max(playerBlindAlpha, (0.55 + 0.45 * distFactor) * facingMul));
        playerBlinded = true;
      }
    }

    Game.snd.flashbang(g.pos, playerBlinded);
  }

  function explodeGrenade(g){
    if (g.kind === 'flash'){
      explodeFlash(g);
      return;
    }
    if (g.kind === 'smoke'){
      Game.snd.smokePop(g.pos);
      spawnSmokeCloud(g.pos);
      return;
    }
    if (g.kind === 'molotov'){
      Game.snd.molotovBurst(g.pos);
      spawnExplosion(g.pos, 0xff5500, 7);
      spawnFireZone(g.pos, g.owner);
      return;
    }

    spawnExplosion(g.pos, 0xff8822, 12);
    Game.snd.explosion(g.pos);

    for (const e of Game.allEntities()){
      if (e.dead) continue;
      const d = Game.dist2D(e.pos, g.pos);
      if (d > Game.CFG.GRENADE_RADIUS) continue;
      if (!Game.grenadeLOS(g.pos, e)) continue;
      const dmg = Game.CFG.GRENADE_MAX_DMG * (1 - d / Game.CFG.GRENADE_RADIUS);
      if (dmg < 2) continue;
      const ph = Game.state.phase;
      if (ph === 'live' || ph === 'roundEnd'){
        Game.applyDamage(e, dmg, g.owner, false, null, 'grenade');
      }
    }
  }

  Game.updateExplosions = function(dt){
    for (let i = explosions.length - 1; i >= 0; i--){
      const e = explosions[i];
      e.life -= dt;
      const k = 1 - e.life / e.max;
      e.s.material.opacity = Math.max(0, 1 - k);
      const sc = 1 + k * (e.maxScale || 12);
      e.s.scale.set(sc, sc, 1);
      if (e.life <= 0){
        Game.scene.remove(e.s);
        e.s.material.dispose();
        explosions.splice(i, 1);
      }
    }

    for (let i = Game.smokeClouds.length - 1; i >= 0; i--){
      const sm = Game.smokeClouds[i];
      sm.life -= dt;
      sm.group.rotation.y += dt * 0.25;
      if (sm.life < 2.5){
        sm.mat.opacity = Math.max(0, (sm.life / 2.5) * 0.86);
      }
      if (sm.life <= 0){
        Game.scene.remove(sm.group);
        sm.mat.dispose();
        Game.smokeClouds.splice(i, 1);
      }
    }

    for (let i = fireZones.length - 1; i >= 0; i--){
      const fz = fireZones[i];
      fz.life -= dt;
      fz.tickTimer -= dt;

      for (let j = 0; j < fz.flames.length; j++){
        fz.flames[j].scale.y = 0.7 + Math.sin(performance.now() * 0.015 + j) * 0.35;
      }

      if (fz.life < 1.5){
        const fade = Math.max(0, fz.life / 1.5);
        fz.baseMat.opacity = fade * 0.55;
        fz.flameMat.opacity = fade * 0.78;
      }

      if (fz.tickTimer <= 0){
        fz.tickTimer = 0.25;
        const tickDmg = Game.CFG.MOLOTOV_DPS * 0.25;
        for (const e of Game.allEntities()){
          if (e.dead) continue;
          const dist = Game.dist2D(e.pos, fz.pos);
          const dy = Math.abs(e.pos.y - fz.pos.y);
          if (dist <= fz.radius && dy < 1.2){
            Game.applyDamage(e, tickDmg, fz.owner, false, null, 'molotov');
          }
        }
      }

      if (fz.life <= 0){
        Game.scene.remove(fz.group);
        fz.baseMat.dispose();
        fz.flameMat.dispose();
        fireZones.splice(i, 1);
      }
    }

    if (playerBlindAlpha > 0){
      const now = performance.now();
      if (now > (Game.player.blindUntil || 0)){
        playerBlindAlpha = Math.max(0, playerBlindAlpha - dt * 0.9);
      }
      Game.$('flashOverlay').style.opacity = playerBlindAlpha.toFixed(3);
    }
  };

  Game.updateGrenades = function(dt){
    for (let i = grenades.length - 1; i >= 0; i--){
      const g = grenades[i];
      g.fuse -= dt;

      g.vel.y -= 18 * dt;
      g.pos.x += g.vel.x * dt;
      g.pos.y += g.vel.y * dt;
      g.pos.z += g.vel.z * dt;

      const gy = Game.getGroundHeight(g.pos.x, g.pos.z) + 0.12;
      if (g.pos.y < gy){
        g.pos.y = gy;
        if (g.kind === 'molotov'){
          g.fuse = 0;
        } else {
          if (g.vel.y < 0) g.vel.y *= -0.35;
          g.vel.x *= 0.7;
          g.vel.z *= 0.7;
        }
      }

      for (const c of Game.colliders){
        const r = 0.12;
        if (g.pos.x > c.minX - r && g.pos.x < c.maxX + r && g.pos.z > c.minZ - r && g.pos.z < c.maxZ + r){
          if (g.pos.y < c.maxY){
            const dxl = g.pos.x - (c.minX - r), dxr = (c.maxX + r) - g.pos.x;
            const dzl = g.pos.z - (c.minZ - r), dzr = (c.maxZ + r) - g.pos.z;
            const m = Math.min(dxl, dxr, dzl, dzr);
            if (m === dxl){ g.pos.x = c.minX - r; g.vel.x = -Math.abs(g.vel.x) * 0.55; }
            else if (m === dxr){ g.pos.x = c.maxX + r; g.vel.x = Math.abs(g.vel.x) * 0.55; }
            else if (m === dzl){ g.pos.z = c.minZ - r; g.vel.z = -Math.abs(g.vel.z) * 0.55; }
            else { g.pos.z = c.maxZ + r; g.vel.z = Math.abs(g.vel.z) * 0.55; }
          }
        }
      }

      g.mesh.rotation.x += g.spinX * dt;
      g.mesh.rotation.z += g.spinZ * dt;
      g.mesh.position.copy(g.pos);

      if (g.fuse <= 0){
        explodeGrenade(g);
        Game.scene.remove(g.mesh);
        grenades.splice(i, 1);
      }
    }
  };

  Game.clearGrenades = function(){
    for (const g of grenades) Game.scene.remove(g.mesh);
    grenades.length = 0;

    for (const sm of Game.smokeClouds){
      Game.scene.remove(sm.group);
      sm.mat.dispose();
    }
    Game.smokeClouds.length = 0;

    for (const fz of fireZones){
      Game.scene.remove(fz.group);
      fz.baseMat.dispose();
      fz.flameMat.dispose();
    }
    fireZones.length = 0;

    playerBlindAlpha = 0;
    Game.$('flashOverlay').style.opacity = 0;
  };

  Game.throwPlayerUtility = function(kind){
    const p = Game.player;
    if (p.dead || Game.state.phase !== 'live' || p.throwCd > 0) return;

    if (kind === 'he'){
      if (p.grenades <= 0){ Game.showMsg('Нет осколочной гранаты — купите в магазине [B]', 1400); return; }
      p.grenades--;
      Game.radioMsg('Вы', 'Бросаю осколочную!');
    } else if (kind === 'flash'){
      if (p.flashes <= 0){ Game.showMsg('Нет световой гранаты — купите в магазине [B]', 1400); return; }
      p.flashes--;
      Game.radioMsg('Вы', 'Световая пошла!');
    } else if (kind === 'smoke'){
      if (p.smokes <= 0){ Game.showMsg('Нет дымовой гранаты — купите в магазине [B]', 1400); return; }
      p.smokes--;
      Game.radioMsg('Вы', 'Ставлю дым!');
    } else if (kind === 'molotov'){
      if (p.molotovs <= 0){ Game.showMsg('Нет зажигательной гранаты — купите в магазине [B]', 1400); return; }
      p.molotovs--;
      Game.radioMsg('Вы', 'Поджигаю!');
    } else {
      return;
    }

    if (p.ads) Game.exitADS();
    p.throwCd = 0.5;
    Game.input.shooting = false;

    const dir = new THREE.Vector3();
    Game.camera.getWorldDirection(dir);
    dir.y += 0.15;
    dir.normalize();
    Game.spawnGrenade(Game.camera.position.clone(), dir, p, kind);

    // Автоматически возвращаем предыдущее оружие после броска гранаты (как в CS)
    const backSlot = (p.prevSlot && p.slots[p.prevSlot]) ? p.prevSlot : (p.slots.primary ? 'primary' : 'secondary');
    const rem = Game.getOwnedGrenades(p);
    p.activeGrenade = rem.length ? rem[0] : null;
    Game.activateSlot(backSlot);
    Game.updHUD();
  };

  /* ---- Стрельба, нож и перезарядка игрока ---- */
  function fireKnifeSlash(w){
    const p = Game.player;
    const rc = Game.bulletRaycaster;
    rc.setFromCamera(new THREE.Vector2(0, 0), Game.camera);
    rc.far = w.range || 2.5;

    const shootables = [...Game.obstacleMeshes];
    for (const b of Game.bots) if (!b.dead && b.team !== 0) shootables.push(b.body, b.head);
    const hits = rc.intersectObjects(shootables);

    if (hits.length && hits[0].object.userData && hits[0].object.userData.bot){
      const h = hits[0];
      const bot = h.object.userData.bot;
      Game.snd.knifeHit();
      const pFwd = new THREE.Vector2(-Math.sin(p.yaw), -Math.cos(p.yaw));
      const bFwd = new THREE.Vector2(Math.sin(bot.yaw), Math.cos(bot.yaw));
      const isBackstab = pFwd.dot(bFwd) > 0.5;
      const dmg = (isBackstab || h.object.userData.head) ? 130 : w.dmg;
      Game.applyDamage(bot, dmg, p, h.object.userData.head, h.point, 'knife');
    } else {
      Game.snd.knifeMiss();
    }
  }

  function fireSingleBullet(w){
    const p = Game.player;
    const rc = Game.bulletRaycaster;
    let spread = w.spread * (p.onGround ? 1 : 2.2);

    if (Game.isLmg(p.weapon)){
      const stab = Math.min(1, Math.max(0, (burstCount - 2) / 11));
      spread = (0.075 * (1 - stab) + 0.004 * stab) * (p.onGround ? 1 : 2.0);
    }

    if (p.crouching) spread *= 0.65;
    if (p.ads) spread *= 0.6;

    rc.setFromCamera(new THREE.Vector2(
      (Math.random() - 0.5) * spread * 2,
      (Math.random() - 0.5) * spread * 2
    ), Game.camera);
    rc.far = 200;

    const shootables = [...Game.obstacleMeshes];
    for (const b of Game.bots) if (!b.dead && b.team !== 0) shootables.push(b.body, b.head);
    const hits = rc.intersectObjects(shootables);
    const from = Game.camera.position.clone();

    if (hits.length){
      const h = hits[0];
      Game.tracer(from, h.point);
      const ud = h.object.userData;
      if (ud && ud.bot){
        const d = from.distanceTo(h.point);
        const falloff = 1 - Math.min(0.25, d / 160);
        Game.applyDamage(ud.bot, w.dmg * falloff, p, ud.head, h.point);
      }
    } else {
      Game.tracer(from, from.clone().add(rc.ray.direction.clone().multiplyScalar(60)));
    }
  }

  function fireShotgunPellets(w){
    const rc = Game.bulletRaycaster;
    const PELLETS = w.pellets || 8;
    const hitsByBot = new Map();
    const from = Game.camera.position.clone();

    for (let i = 0; i < PELLETS; i++){
      const spread = w.spread * (Game.player.crouching ? 0.75 : 1);
      rc.setFromCamera(new THREE.Vector2(
        (Math.random() - 0.5) * spread * 2,
        (Math.random() - 0.5) * spread * 2
      ), Game.camera);
      rc.far = 60;
      const shootables = [...Game.obstacleMeshes];
      for (const b of Game.bots) if (!b.dead && b.team !== 0) shootables.push(b.body, b.head);
      const hits = rc.intersectObjects(shootables);
      if (hits.length){
        const h = hits[0];
        Game.tracer(from, h.point);
        const ud = h.object.userData;
        if (ud && ud.bot){
          const dist = from.distanceTo(h.point);
          const falloff = Math.max(0.3, 1 - dist / 25);
          const perDmg = w.dmg * falloff;
          const info = hitsByBot.get(ud.bot) || { headDmg: 0, bodyDmg: 0, point: h.point.clone() };
          if (ud.head) info.headDmg += perDmg;
          else info.bodyDmg += perDmg;
          if (!info.point) info.point = h.point.clone();
          hitsByBot.set(ud.bot, info);
        }
      } else {
        Game.tracer(from, from.clone().add(rc.ray.direction.clone().multiplyScalar(30)));
      }
    }

    for (const [victim, info] of hitsByBot){
      applyShotgunDamage(victim, info.headDmg, info.bodyDmg, info.point);
    }
  }

  Game.startReload = function(){
    const p = Game.player;
    if (p.currentSlot === 'grenade') return;
    const w = Game.WEAPONS[p.weapon];
    if (w.melee || p.reloading || p.ammo === w.mag || p.reserve <= 0) return;
    if (p.ads) Game.exitADS();
    burstCount = 0;
    p.reloading = true;
    p.reloadEnd = performance.now() + w.reload * 1000;
    Game.showMsg('Перезарядка...', w.reload * 1000);
  };

  Game.finishReload = function(){
    const p = Game.player;
    const w = Game.WEAPONS[p.weapon];
    const need = w.mag - p.ammo;
    const take = Math.min(need, p.reserve);
    p.ammo += take;
    p.reserve -= take;
    p.reloading = false;
    Game.syncAmmoToSlot();
    Game.updHUD();
  };

  Game.playerShoot = function(){
    const p = Game.player;
    if (p.dead || p.reloading || p.shootCd > 0) return;

    // Если в руках граната (слот [4]) — бросаем её по ЛКМ
    if (p.currentSlot === 'grenade'){
      if (p.activeGrenade) Game.throwPlayerUtility(p.activeGrenade);
      return;
    }

    const w = Game.WEAPONS[p.weapon];

    if (w.melee){
      p.shootCd = w.rate / 1000;
      Game.gunFx.slash = 1.0;
      fireKnifeSlash(w);
      return;
    }

    if (p.ammo <= 0){ Game.startReload(); return; }

    const now = performance.now();
    if (now - lastShotTime > 260) burstCount = 0;
    burstCount++;
    lastShotTime = now;

    p.ammo--;
    p.shootCd = w.rate / 1000;

    if (w.pellets){
      Game.snd.shotgun();
      Game.triggerMuzzleFlash(0.06);
      Game.gunFx.recoil = Math.min(1.0, Game.gunFx.recoil + 1.2);
      fireShotgunPellets(w);
    } else {
      if (Game.isSniper(p.weapon)) Game.snd.sniper(); else Game.snd.shot();
      Game.triggerMuzzleFlash(0.05);
      Game.gunFx.recoil = Math.min(0.9, Game.gunFx.recoil + (Game.isSniper(p.weapon) ? 1.0 : 0.55));
      fireSingleBullet(w);
    }

    let kick = 0.006;
    if (Game.isSniper(p.weapon)) kick = 0.02;
    else if (w.pellets) kick = 0.03;
    else if (Game.isLmg(p.weapon)) kick = burstCount > 10 ? 0.0015 : 0.009;
    if (p.crouching) kick *= 0.7;

    p.pitch += kick + Math.random() * 0.003;
    p.yaw += (Math.random() - 0.5) * (Game.isLmg(p.weapon) && burstCount > 10 ? 0.001 : 0.004);
    Game.syncAmmoToSlot();
    Game.updHUD();
  };

  function isPathClear(ax, az, bx, bz){
    const ay = Game.getGroundHeight(ax, az) + 0.9;
    const by = Game.getGroundHeight(bx, bz) + 0.9;
    const origin = new THREE.Vector3(ax, ay, az);
    const dir = new THREE.Vector3(bx - ax, by - ay, bz - az);
    const len = dir.length();
    if (len < 0.2) return true;
    Game.raycaster.set(origin, dir.normalize());
    Game.raycaster.far = len;
    return Game.raycaster.intersectObjects(Game.obstacleMeshes).length === 0;
  }

  function pickNewWaypoint(bot, goalPos){
    const wp = Game.wayPoints;
    let candidates = wp.filter(w => {
      const d = Math.hypot(w.x - bot.pos.x, w.z - bot.pos.z);
      return d > 4 && d < 28 && isPathClear(bot.pos.x, bot.pos.z, w.x, w.z);
    });

    if (!candidates.length){
      candidates = wp.filter(w => Math.hypot(w.x - bot.pos.x, w.z - bot.pos.z) > 5);
    }
    const pool = candidates.length ? candidates : wp;

    if (goalPos){
      let best = pool[0];
      let bestScore = 1e9;
      for (const w of pool){
        const distToGoal = Math.hypot(w.x - goalPos.x, w.z - goalPos.z);
        const score = distToGoal + Math.random() * 4;
        if (score < bestScore){
          bestScore = score;
          best = w;
        }
      }
      return best;
    }

    return pool[Math.floor(Math.random() * pool.length)];
  }

  function findNearestEnemy(bot){
    let nearest = null;
    let bestD = 1e9;
    for (const e of Game.allEntities()){
      if (e.team === bot.team || e.dead) continue;
      const d = Game.dist2D(bot.pos, e.pos);
      if (d < bestD){
        bestD = d;
        nearest = e;
      }
    }
    return nearest;
  }

  function findVisibleEnemy(bot){
    let best = null, bestD = 1e9;
    for (const e of Game.allEntities()){
      if (e.team === bot.team || e.dead) continue;
      const d = Game.dist2D(bot.pos, e.pos);
      if (d < bestD && d < 45 && Game.hasLOS(bot, e)){
        best = e;
        bestD = d;
      }
    }
    return { enemy: best, dist: bestD };
  }

  function findCoverPoint(bot, threatPos){
    const ARENA = Game.CFG.ARENA;
    let bestPt = null;
    let bestScore = 1e9;

    for (const c of Game.colliders){
      if (c.maxY < 1.2) continue;
      const w = c.maxX - c.minX;
      const d = c.maxZ - c.minZ;
      if (w > 20 || d > 20) continue;

      const cx = (c.minX + c.maxX) * 0.5;
      const cz = (c.minZ + c.maxZ) * 0.5;
      let dx = cx - threatPos.x;
      let dz = cz - threatPos.z;
      const len = Math.hypot(dx, dz) || 1;
      dx /= len;
      dz /= len;

      const offset = Math.max(w, d) * 0.5 + 1.15;
      const px = Math.max(-ARENA + 1.2, Math.min(ARENA - 1.2, cx + dx * offset));
      const pz = Math.max(-ARENA + 1.2, Math.min(ARENA - 1.2, cz + dz * offset));

      const distToBot = Math.hypot(px - bot.pos.x, pz - bot.pos.z);
      const distToThreat = Math.hypot(px - threatPos.x, pz - threatPos.z);
      const score = distToBot - distToThreat * 0.35 + Math.random() * 3.0;
      if (score < bestScore){
        bestScore = score;
        bestPt = new THREE.Vector3(px, 0, pz);
      }
    }
    return bestPt;
  }

  Game.botBuyRound = function(bot){
    const W = Game.WEAPONS;
    const CFG = Game.CFG;
    const curTier = WEAPON_TIER[bot.weapon] || 0;

    if (bot.armor < 100 || !bot.helmet){
      const upgradeCost = (bot.armor >= 100 && !bot.helmet) ? CFG.HELMET_UPGRADE_PRICE : CFG.HELMET_ARMOR_PRICE;
      if (bot.money >= upgradeCost + 1200 || (curTier >= 3 && bot.money >= upgradeCost)){
        bot.money -= upgradeCost;
        bot.armor = 100;
        bot.helmet = true;
      } else if (bot.armor < 100 && bot.money >= CFG.ARMOR_PRICE){
        bot.money -= CFG.ARMOR_PRICE;
        bot.armor = 100;
      }
    }

    if (bot.helmetMesh) bot.helmetMesh.visible = Boolean(bot.helmet && bot.armor > 0);

    if (!Game.isAttackTeam(bot.team) && !bot.defuseKit && bot.money >= CFG.DEFUSE_KIT_PRICE + 600 && Math.random() < 0.65){
      bot.money -= CFG.DEFUSE_KIT_PRICE;
      bot.defuseKit = true;
    }

    const candidates = [
      { key: 'pistol_mid',    weight: 1.0 },
      { key: 'pistol_deagle', weight: 1.4 },
      { key: 'smg',           weight: 1.8 },
      { key: 'negev',         weight: 1.7 },
      { key: 'shotgun',       weight: 1.8 },
      { key: 'rifle',         weight: 2.8 },
      { key: 'sniper_light',  weight: 2.0 },
      { key: 'sniper_heavy',  weight: 2.2 }
    ].filter(item => {
      const w = W[item.key];
      return w && WEAPON_TIER[item.key] > curTier && bot.money >= w.price;
    });

    if (candidates.length > 0){
      let pool = candidates;
      if (bot.money >= 2500){
        const primaries = candidates.filter(c => WEAPON_TIER[c.key] >= 3);
        if (primaries.length > 0) pool = primaries;
      }

      const totalWeight = pool.reduce((sum, c) => sum + c.weight, 0);
      let roll = Math.random() * totalWeight;
      let chosen = pool[pool.length - 1].key;
      for (const c of pool){
        roll -= c.weight;
        if (roll <= 0){
          chosen = c.key;
          break;
        }
      }

      bot.money -= W[chosen].price;
      bot.weapon = chosen;
    }

    if (bot.money >= CFG.GRENADE_PRICE && (bot.grenades || 0) < 1 && Math.random() < 0.7){
      bot.money -= CFG.GRENADE_PRICE;
      bot.grenades = 1;
    }
    if (bot.money >= CFG.FLASH_PRICE && (bot.flashes || 0) < 1 && Math.random() < 0.7){
      bot.money -= CFG.FLASH_PRICE;
      bot.flashes = 1;
    }
    if (bot.money >= CFG.SMOKE_PRICE && (bot.smokes || 0) < 1 && Math.random() < 0.7){
      bot.money -= CFG.SMOKE_PRICE;
      bot.smokes = 1;
    }
    if (bot.money >= CFG.MOLOTOV_PRICE && (bot.molotovs || 0) < 1 && Math.random() < 0.65){
      bot.money -= CFG.MOLOTOV_PRICE;
      bot.molotovs = 1;
    }

    const wObj = W[bot.weapon];
    bot.ammo = wObj.mag;
    bot.reserve = wObj.mag * 3;
    bot.reloading = false;
    bot.reloadEnd = 0;

    setBotWeapon(bot, bot.weapon);
  };

  function botShoot(bot, target){
    const w = Game.WEAPONS[bot.weapon];
    if (bot.reloading) return;
    if (bot.ammo <= 0){
      bot.reloading = true;
      bot.reloadEnd = performance.now() + w.reload * 1000;
      return;
    }
    bot.ammo--;

    const d = Game.dist2D(bot.pos, target.pos);
    const isSnip = Game.isSniper(bot.weapon);
    const isShot = bot.weapon === 'shotgun';
    const isLmg  = Game.isLmg(bot.weapon);

    let baseAcc;
    if (isSnip){
      baseAcc = Math.max(0.32, 0.58 - d * 0.003);
    } else if (isShot){
      baseAcc = Math.max(0.12, 0.72 - d * 0.025);
    } else if (isLmg){
      baseAcc = Math.max(0.20, 0.46 - d * 0.004);
    } else {
      baseAcc = Math.max(0.15, 0.42 - d * 0.0045);
    }

    const chance = baseAcc * (bot.crouching ? 1.15 : 1) * (bot.hp < 40 ? 0.85 : 1) * (target.isPlayer ? 0.95 : 1);

    const willHit = Math.random() < chance;
    const isHead  = willHit && Math.random() < (isSnip ? 0.25 : (isShot ? 0.15 : 0.33));
    const fromY = (bot.pos.y || 0) + (bot.crouching ? 1.15 : 1.6);
    const from = new THREE.Vector3(bot.pos.x, fromY, bot.pos.z);
    const targetY = (target.pos.y || 0) + (target.crouching ? (isHead ? 1.35 : 0.95) : (isHead ? 1.85 : 1.3));
    const to = new THREE.Vector3(target.pos.x, targetY, target.pos.z);
    const aim = to.clone().sub(from).normalize();

    if (!willHit){
      aim.add(new THREE.Vector3((Math.random() - 0.5) * 0.35, (Math.random() - 0.2) * 0.2, (Math.random() - 0.5) * 0.35));
      aim.normalize();
    }

    Game.raycaster.set(from, aim);
    Game.raycaster.far = 70;
    const hits = Game.raycaster.intersectObjects(Game.obstacleMeshes);
    let endPoint = from.clone().add(aim.clone().multiplyScalar(60));
    if (hits.length) endPoint = hits[0].point;
    Game.tracer(from, endPoint);
    Game.snd.botShot(from, bot.weapon);

    if (willHit){
      const dist = from.distanceTo(to);
      if (!hits.length || from.distanceTo(endPoint) > dist - 0.5){
        let dmg = w.dmg * (0.85 + Math.random() * 0.3);
        if (isShot){
          const pelletsHit = Math.max(2, Math.round(6 * Math.max(0.25, 1 - d / 24)));
          dmg *= pelletsHit;
        }
        Game.applyDamage(target, dmg, bot, isHead, to);
        if (target.isPlayer) Game.snd.hit();
      }
    }

    if (bot.ammo <= 0){
      bot.reloading = true;
      bot.reloadEnd = performance.now() + w.reload * 1000;
    }
  }

  function avoidObstacles(bot, move, dt){
    if (move.lengthSq() < 0.001) return;
    if (bot.avoidTimer > 0) bot.avoidTimer -= dt;

    const origin = new THREE.Vector3(bot.pos.x, (bot.pos.y || 0) + 0.9, bot.pos.z);
    const dir = move.clone().normalize();
    Game.raycaster.set(origin, dir);
    Game.raycaster.far = 2.4;

    if (Game.raycaster.intersectObjects(Game.obstacleMeshes).length > 0){
      if (bot.avoidTimer <= 0){
        bot.avoidDir = (bot.pos.z >= 0) ? -1 : 1;
        bot.avoidTimer = 1.6;
      }

      const d1 = bot.avoidDir;
      const d2 = -d1;
      const angles = [
        d1 * Math.PI / 6,
        d1 * Math.PI / 3,
        d1 * Math.PI / 2,
        d1 * (2 * Math.PI / 3),
        d2 * Math.PI / 3,
        d2 * Math.PI / 2,
        d2 * (2 * Math.PI / 3),
        d1 * (5 * Math.PI / 6)
      ];

      const baseX = dir.x, baseZ = dir.z;
      for (const a of angles){
        const c = Math.cos(a), s = Math.sin(a);
        const tryX = baseX * c - baseZ * s;
        const tryZ = baseX * s + baseZ * c;
        Game.raycaster.set(origin, new THREE.Vector3(tryX, 0, tryZ));
        Game.raycaster.far = 2.2;
        if (Game.raycaster.intersectObjects(Game.obstacleMeshes).length === 0){
          move.set(tryX, 0, tryZ);
          return;
        }
      }
    }
  }

  function applyBotPose(bot){
    bot.pos.y = Game.getGroundHeight(bot.pos.x, bot.pos.z);
    bot.mesh.position.copy(bot.pos);
    bot.mesh.rotation.y = bot.yaw;
    bot.mesh.scale.y = bot.crouching ? 0.7 : 1.0;
  }

  Game.updateBot = function(bot, dt){
    if (bot.dead) return;
    const now = performance.now();
    const isBlinded = now < (bot.blindUntil || 0);
    const bomb = Game.bomb;
    const isAttack = Game.isAttackTeam(bot.team);

    // Отслеживаем получение урона в этом кадре
    if (bot._lastHp === undefined || bot.hp > bot._lastHp) bot._lastHp = bot.hp;
    const tookHit = bot.hp < bot._lastHp;
    if (tookHit) bot._lastHp = bot.hp;

    const carriesBomb = bomb.status === 'carried' && bomb.carrier === bot;
    if (bot.c4BeltMesh){
      bot.c4BeltMesh.visible = carriesBomb;
      if (carriesBomb && bot.c4Led){
        bot.c4Led.visible = Math.floor(now / 350) % 2 === 0;
      }
    }

    if (bot.reloading && now >= bot.reloadEnd){
      const w = Game.WEAPONS[bot.weapon];
      if (bot.reserve > 0){
        const take = Math.min(w.mag - bot.ammo, bot.reserve);
        bot.ammo += take;
        bot.reserve -= take;
      } else {
        bot.ammo = w.mag;
      }
      bot.reloading = false;
    }

    if (isBlinded){
      if (bomb.actionActor === bot && Game.cancelBombAction) Game.cancelBombAction(bot);
      bot.crouching = false;
      bot.target = null;
      bot.yaw += dt * 2.5 * bot.avoidDir;
      const blindMove = new THREE.Vector3(-Math.sin(bot.yaw), 0, -Math.cos(bot.yaw));
      avoidObstacles(bot, blindMove, dt);
      bot.pos.x += blindMove.x * 2.1 * dt;
      bot.pos.z += blindMove.z * 2.1 * dt;
      Game.resolveCollisions(bot.pos, 0.5);
      applyBotPose(bot);
      return;
    }

    // Проверка на прерывание закладки/разминирования при получении урона или обнаружении врага
    const wantsPlant = carriesBomb && Game.getSiteAt(bot.pos) && now >= (bot.plantAbortUntil || 0);
    const wantsDefuse = bomb.status === 'planted' && !isAttack && Game.dist2D(bot.pos, bomb.pos) <= Game.CFG.BOMB_USE_RADIUS && now >= (bot.plantAbortUntil || 0);

    if (wantsPlant || wantsDefuse){
      const vis = findVisibleEnemy(bot);
      if (tookHit || vis.enemy){
        // Прерываем работу с бомбой, открываем огонь и уходим с линии обстрела в укрытие
        if (bomb.actionActor === bot && Game.cancelBombAction) Game.cancelBombAction(bot);
        bot.plantAbortUntil = now + 4500;
        bot.evadeUntil = now + 2600;
        bot.crouching = false;
        const threat = vis.enemy || findNearestEnemy(bot);
        if (vis.enemy){
          bot.target = vis.enemy;
          bot.aimT = bot.reaction; // мгновенная реакция, чтобы сразу отстреливаться
        }
        if (threat){
          bot.coverPoint = findCoverPoint(bot, threat.pos);
          bot.coverTimer = 2.6;
        }
      } else {
        bot.crouching = true;
        Game.advanceBombAction(bot, wantsPlant ? 'plant' : 'defuse', dt);
        applyBotPose(bot);
        return;
      }
    } else if (bomb.actionActor === bot && Game.cancelBombAction){
      Game.cancelBombAction(bot);
    }

    const endGameRush = Game.state.roundTimeLeft <= 25;
    const evadingFire = now < (bot.evadeUntil || 0);
    const lowHp = (!endGameRush && bot.hp <= 38) || evadingFire;

    if (bot.coverTimer > 0) bot.coverTimer -= dt;

    bot.think -= dt;
    if (bot.think <= 0 || tookHit){
      bot.think = 0.25 + Math.random() * 0.2;
      const vis = findVisibleEnemy(bot);
      const best = vis.enemy;
      const bestD = vis.dist;

      if (best !== bot.target){
        bot.target = best;
        bot.aimT = tookHit ? bot.reaction : 0;
        bot.reaction = Game.isSniper(bot.weapon) ? (0.35 + Math.random() * 0.25) : (0.2 + Math.random() * 0.2);
      }

      bot.crouching = Boolean(bot.target && !evadingFire && bestD > 8 && Math.random() < 0.35);

      if (lowHp && (bot.target || tookHit)){
        const threat = bot.target || findNearestEnemy(bot);
        if (threat && (!bot.coverPoint || bot.coverTimer <= 0)){
          bot.coverPoint = findCoverPoint(bot, threat.pos);
          bot.coverTimer = 2.5 + Math.random() * 1.5;
        }
      } else if (!evadingFire){
        bot.coverPoint = null;
      }

      let objectiveGoal = null;
      if (bomb.status === 'planted'){
        objectiveGoal = bomb.pos;
      } else if (isAttack){
        if (bomb.status === 'dropped'){
          objectiveGoal = bomb.pos;
        } else if (carriesBomb){
          if (!bot.chosenSite){
            const sites = Game.CFG.BOMBSITES;
            bot.chosenSite = sites[Math.floor(Math.random() * sites.length)];
          }
          objectiveGoal = new THREE.Vector3(bot.chosenSite.x, 0, bot.chosenSite.z);
        }
      }

      if (!best && !bot.coverPoint){
        if (objectiveGoal){
          if (isPathClear(bot.pos.x, bot.pos.z, objectiveGoal.x, objectiveGoal.z)){
            bot.waypoint = objectiveGoal.clone();
          } else if (!bot.waypoint || Game.dist2D(bot.pos, bot.waypoint) < 2.2 || !isPathClear(bot.pos.x, bot.pos.z, bot.waypoint.x, bot.waypoint.z)){
            bot.waypoint = pickNewWaypoint(bot, objectiveGoal);
          }
        } else if (endGameRush){
          const enemy = findNearestEnemy(bot);
          if (enemy){
            if (isPathClear(bot.pos.x, bot.pos.z, enemy.pos.x, enemy.pos.z)){
              bot.waypoint = enemy.pos.clone();
            } else if (!bot.waypoint || Game.dist2D(bot.pos, bot.waypoint) < 2.2 || !isPathClear(bot.pos.x, bot.pos.z, bot.waypoint.x, bot.waypoint.z)){
              bot.waypoint = pickNewWaypoint(bot, enemy.pos);
            }
          } else if (!bot.waypoint || Game.dist2D(bot.pos, bot.waypoint) < 2.5){
            bot.waypoint = pickNewWaypoint(bot);
          }
        } else if (!bot.waypoint || Game.dist2D(bot.pos, bot.waypoint) < 2.5){
          bot.waypoint = pickNewWaypoint(bot);
        }
      }
    }

    const move = new THREE.Vector3();
    const isSnip = Game.isSniper(bot.weapon);
    const isShot = bot.weapon === 'shotgun';
    const isLmg  = Game.isLmg(bot.weapon);

    bot.strafeT -= dt;
    if (bot.strafeT <= 0){
      bot.strafeDir = Math.random() < 0.5 ? -1 : 1;
      bot.strafeT = 0.8 + Math.random() * 1.4;
    }

    if (lowHp && bot.coverPoint){
      bot.crouching = false;
      const toCover = new THREE.Vector3(bot.coverPoint.x - bot.pos.x, 0, bot.coverPoint.z - bot.pos.z);
      const distToCover = toCover.length();

      if (distToCover > 1.1){
        move.copy(toCover).normalize();
        avoidObstacles(bot, move, dt);
      } else {
        bot.crouching = !evadingFire;
        const refPos = bot.target ? bot.target.pos : bot.coverPoint;
        const dir = new THREE.Vector3(refPos.x - bot.pos.x, 0, refPos.z - bot.pos.z).normalize();
        move.set(-dir.z * bot.strafeDir, 0, dir.x * bot.strafeDir);
        avoidObstacles(bot, move, dt);
      }

      if (bot.target){
        const t = bot.target;
        const dir = new THREE.Vector3(t.pos.x - bot.pos.x, 0, t.pos.z - bot.pos.z).normalize();
        bot.yaw = Math.atan2(dir.x, dir.z);
        bot.aimT += dt;
        bot.shootCd -= dt;
        if (bot.shootCd <= 0 && bot.aimT >= bot.reaction){
          bot.shootCd = (Game.WEAPONS[bot.weapon].rate / 1000) * (1.3 + Math.random() * 1.0);
          botShoot(bot, t);
        }
      } else if (move.lengthSq() > 0.01){
        bot.yaw = Math.atan2(move.x, move.z);
      }
    } else if (bot.target){
      const t = bot.target;
      const d = Game.dist2D(bot.pos, t.pos);
      const dir = new THREE.Vector3(t.pos.x - bot.pos.x, 0, t.pos.z - bot.pos.z).normalize();
      bot.yaw = Math.atan2(dir.x, dir.z);

      const minRange = isSnip ? 14 : (isShot ? 3 : 6);
      const maxRange = isSnip ? 28 : (isShot ? 9 : 16);

      if (d > maxRange) move.copy(dir);
      else if (d < minRange) move.copy(dir).multiplyScalar(-1);
      else move.set(-dir.z * bot.strafeDir, 0, dir.x * bot.strafeDir);

      avoidObstacles(bot, move, dt);

      bot.aimT += dt;
      bot.shootCd -= dt;
      if (bot.shootCd <= 0 && bot.aimT >= bot.reaction){
        const rateMul = isLmg ? (1.05 + Math.random() * 0.45) : (1.3 + Math.random() * 1.1);
        bot.shootCd = (Game.WEAPONS[bot.weapon].rate / 1000) * rateMul;
        botShoot(bot, t);
      }

      const hasActiveGr = Game.grenades.some(gr => gr.owner === bot);
      if (!hasActiveGr && d > 7 && d < 24 && Math.random() < 0.014){
        const origin = new THREE.Vector3(bot.pos.x, (bot.pos.y || 0) + 1.6, bot.pos.z);
        const tdir = new THREE.Vector3(t.pos.x - bot.pos.x, 0, t.pos.z - bot.pos.z).normalize();
        tdir.y = 0.26;
        tdir.normalize();

        if (bot.molotovs > 0 && d > 8 && d < 20){
          bot.molotovs--;
          Game.spawnGrenade(origin, tdir, bot, 'molotov');
          if (bot.team === Game.player.team) Game.radioMsg(bot.name, 'Поджигаю!');
        } else if (bot.flashes > 0 && d < 18){
          bot.flashes--;
          Game.spawnGrenade(origin, tdir, bot, 'flash');
          if (bot.team === Game.player.team) Game.radioMsg(bot.name, 'Световая пошла!');
        } else if (bot.smokes > 0 && (bot.hp < 55 || isSnip || isLmg)){
          bot.smokes--;
          Game.spawnGrenade(origin, tdir, bot, 'smoke');
          if (bot.team === Game.player.team) Game.radioMsg(bot.name, 'Ставлю дым!');
        } else if (bot.grenades > 0 && d > 8 && d < 22){
          bot.grenades--;
          Game.spawnGrenade(origin, tdir, bot, 'he');
          if (bot.team === Game.player.team) Game.radioMsg(bot.name, 'Бросаю осколочную!');
        }
      }
    } else if (bot.waypoint){
      bot.crouching = false;
      bot.aimT = 0;
      move.set(bot.waypoint.x - bot.pos.x, 0, bot.waypoint.z - bot.pos.z).normalize();
      avoidObstacles(bot, move, dt);
      bot.yaw = Math.atan2(move.x, move.z);
    }

    const wObj = Game.WEAPONS[bot.weapon];
    const botSpd = 5.4 * (wObj.speedMul || 1) * (bot.crouching ? 0.5 : 1) * (now < (bot.slowUntil || 0) ? 0.45 : 1);
    const beforeX = bot.pos.x, beforeZ = bot.pos.z;
    bot.pos.x += move.x * botSpd * dt;
    bot.pos.z += move.z * botSpd * dt;
    Game.resolveCollisions(bot.pos, 0.5);

    const movedDist = Math.hypot(bot.pos.x - beforeX, bot.pos.z - beforeZ);
    const wantedDist = Math.hypot(move.x, move.z) * botSpd * dt;

    if (wantedDist > 0.005 && movedDist < wantedDist * 0.4){
      bot.stuckTimer = (bot.stuckTimer || 0) + dt;
      const perpX = -move.z, perpZ = move.x;
      const perpLen = Math.hypot(perpX, perpZ);
      if (perpLen > 0.001){
        const step = botSpd * dt * 1.3;
        bot.pos.x += (perpX / perpLen) * step * bot.avoidDir;
        bot.pos.z += (perpZ / perpLen) * step * bot.avoidDir;
        Game.resolveCollisions(bot.pos, 0.5);
      }
      if (bot.stuckTimer > 0.7){
        bot.avoidDir *= -1;
        bot.avoidTimer = 1.6;
        if (bot.coverPoint) bot.coverPoint = null;
        if (!bot.target) bot.waypoint = pickNewWaypoint(bot);
        bot.stuckTimer = 0;
      }
    } else {
      bot.stuckTimer = 0;
    }

    if (move.lengthSq() > 0.01 && !bot.crouching){
      bot.stepTimer -= dt;
      if (bot.stepTimer <= 0){
        bot.stepTimer = 0.34;
        Game.snd.footstep(bot.pos, 1);
      }
    } else {
      bot.stepTimer = 0;
    }

    for (const o of Game.allEntities()){
      if (o === bot || o.dead) continue;
      const dx = bot.pos.x - o.pos.x, dz = bot.pos.z - o.pos.z;
      const dd = Math.hypot(dx, dz);
      if (dd < 1 && dd > 0.01){ bot.pos.x += dx / dd * 0.5 * dt * 4; bot.pos.z += dz / dd * 0.5 * dt * 4; }
    }
    applyBotPose(bot);
  };
})();'use strict';

(function(){
  Game.spectateList = function(){
    let list = Game.bots.filter(b => !b.dead && b.team === Game.player.team);
    if (!list.length) list = Game.bots.filter(b => !b.dead);
    return list;
  };

  Game.cycleCamMode = function(){
    const st = Game.state;
    st.camMode = (st.camMode + 1) % 3;
    st.specYawOffset = 0;
    st.specPitchOffset = 0;
    if (st.camMode === 2){
      st.freeCamPos.copy(Game.camera.position);
      const dir = new THREE.Vector3();
      Game.camera.getWorldDirection(dir);
      st.freeCamPitch = Math.asin(Math.max(-1, Math.min(1, dir.y)));
      st.freeCamYaw = Math.atan2(-dir.x, -dir.z);
    }
  };

  function updateThirdPersonCam(dt){
    const st = Game.state;
    const list = Game.spectateList();
    if (!list.length) return;
    if (st.specIdx >= list.length) st.specIdx = 0;
    const b = list[st.specIdx];
    if (!b || b.dead) return;

    const totalYaw = b.yaw + st.specYawOffset;
    const pitch = Math.max(-0.7, Math.min(1.0, st.specPitchOffset));
    const dist = 3.5;
    const horiz = Math.cos(pitch) * dist;
    const vert  = Math.sin(pitch) * dist;
    const camX = b.pos.x - Math.sin(totalYaw) * horiz;
    const camY = b.pos.y + 1.4 + vert;
    const camZ = b.pos.z - Math.cos(totalYaw) * horiz;

    const from = new THREE.Vector3(b.pos.x, b.pos.y + 1.4, b.pos.z);
    const to = new THREE.Vector3(camX, camY, camZ);
    const dir = to.clone().sub(from);
    const len = dir.length();
    let finalX = camX, finalY = camY, finalZ = camZ;

    if (len > 0.001){
      Game.raycaster.set(from, dir.clone().normalize());
      Game.raycaster.far = len;
      const hits = Game.raycaster.intersectObjects(Game.obstacleMeshes);
      if (hits.length > 0){
        const hp = hits[0].point;
        const d = Math.max(0.5, hp.distanceTo(from) - 0.3);
        const nd = dir.clone().normalize().multiplyScalar(d);
        finalX = from.x + nd.x;
        finalY = from.y + nd.y;
        finalZ = from.z + nd.z;
      }
    }

    Game.camera.position.lerp(new THREE.Vector3(finalX, finalY, finalZ), Math.min(1, dt * 8));
    Game.camera.lookAt(b.pos.x, b.pos.y + 1.4, b.pos.z);

    Game.specBarShow('👁 3-е лицо: ' + b.name + ' · V — режим · ЛКМ — след. игрок');
  }

  function updateFPVCam(){
    const st = Game.state;
    const list = Game.spectateList();
    if (!list.length) return;
    if (st.specIdx >= list.length) st.specIdx = 0;
    const b = list[st.specIdx];
    if (!b || b.dead){ st.specIdx = (st.specIdx + 1) % Math.max(1, list.length); return; }

    const totalYaw = b.yaw + st.specYawOffset;
    const pitch = Math.max(-1.4, Math.min(1.4, st.specPitchOffset));
    const lookDist = 6;
    const lookX = b.pos.x + Math.sin(totalYaw) * Math.cos(pitch) * lookDist;
    const lookY = b.pos.y + 1.6 + Math.sin(pitch) * lookDist;
    const lookZ = b.pos.z + Math.cos(totalYaw) * Math.cos(pitch) * lookDist;

    Game.camera.position.set(b.pos.x, b.pos.y + 1.6, b.pos.z);
    Game.camera.lookAt(lookX, lookY, lookZ);

    Game.specBarShow('👁 1-е лицо: ' + b.name + ' · V — режим · ЛКМ — след. игрок');
  }

  function updateFreeCam(dt){
    const st = Game.state;
    const keys = Game.keys;
    const ARENA = Game.CFG.ARENA;
    const speed = keys.has('ShiftLeft') ? 15 : 8;
    const f = new THREE.Vector3(-Math.sin(st.freeCamYaw), 0, -Math.cos(st.freeCamYaw));
    const r = new THREE.Vector3( Math.cos(st.freeCamYaw), 0, -Math.sin(st.freeCamYaw));
    const move = new THREE.Vector3();
    if (keys.has('KeyW')) move.add(f);
    if (keys.has('KeyS')) move.sub(f);
    if (keys.has('KeyD')) move.add(r);
    if (keys.has('KeyA')) move.sub(r);
    if (keys.has('Space')) move.y += 1;
    if (keys.has('KeyC')) move.y -= 1;
    if (move.lengthSq() > 0) move.normalize().multiplyScalar(speed * dt);
    st.freeCamPos.add(move);
    st.freeCamPos.x = Math.max(-ARENA + 1, Math.min(ARENA - 1, st.freeCamPos.x));
    st.freeCamPos.z = Math.max(-ARENA + 1, Math.min(ARENA - 1, st.freeCamPos.z));
    st.freeCamPos.y = Math.max(0.5, Math.min(25, st.freeCamPos.y));

    Game.camera.position.copy(st.freeCamPos);
    Game.camera.rotation.y = st.freeCamYaw;
    Game.camera.rotation.x = st.freeCamPitch;

    Game.specBarShow('👁 Свободная камера · WASD+Space/C — полёт · V — режим');
  }

  Game.updateSpectator = function(dt){
    const st = Game.state;
    for (const b of Game.bots){
      if (!b.dead) b.mesh.visible = true;
    }

    if (st.camMode === 2){ updateFreeCam(dt); return; }
    if (st.camMode === 1){
      const list = Game.spectateList();
      if (!list.length){ updateFreeCam(dt); return; }
      if (st.specIdx >= list.length) st.specIdx = 0;
      const b = list[st.specIdx];
      if (!b){ updateFreeCam(dt); return; }
      b.mesh.visible = false;
      updateFPVCam();
      return;
    }
    updateThirdPersonCam(dt);
  };

  function updatePlayer(dt){
    const p = Game.player;
    const st = Game.state;
    const keys = Game.keys;
    const bomb = Game.bomb;
    if (p.dead){ Game.updateSpectator(dt); return; }

    const isGrenadeSlot = p.currentSlot === 'grenade';
    const w = isGrenadeSlot ? { rate: 500, speedMul: 1.05 } : Game.WEAPONS[p.weapon];
    if (p.reloading && performance.now() >= p.reloadEnd) Game.finishReload();
    p.shootCd -= dt;
    if (p.throwCd > 0) p.throwCd -= dt;

    // Приседание перенесено на [C], чтобы избежать конфликтов с Ctrl-комбинациями браузера
    const wantCrouch = keys.has('KeyC') && !st.menuOpen;
    p.crouching = wantCrouch;
    const targetEye = p.crouching ? 1.15 : 1.7;
    p.eyeHeight += (targetEye - p.eyeHeight) * Math.min(1, dt * 14);

    let doingBombAction = false;
    let lookAtWeapon = false;

    if (st.phase === 'live' && !st.menuOpen){
      const canPlant = bomb.status === 'carried' && bomb.carrier === p && Game.getSiteAt(p.pos);
      const canDefuse = bomb.status === 'planted' && !Game.isAttackTeam(p.team) && Game.dist2D(p.pos, bomb.pos) <= CFG.BOMB_USE_RADIUS;

      const origin = new THREE.Vector3(p.pos.x, p.pos.y + p.eyeHeight, p.pos.z);
      const dir = new THREE.Vector3(-Math.sin(p.yaw), -Math.sin(p.pitch), -Math.cos(p.yaw)).normalize();

      for (const dw of Game.droppedWeapons){
        if (origin.distanceTo(dw.pos) < 3.5 && dir.dot(dw.pos.clone().sub(origin).normalize()) > 0.8){
          lookAtWeapon = true;
          break;
        }
      }

      if (keys.has('KeyE') && (canPlant || canDefuse)){
        doingBombAction = true;
        p.crouching = true;
        if (p.ads) Game.exitADS();
        Game.advanceBombAction(p, canPlant ? 'plant' : 'defuse', dt);
      } else {
        if (bomb.actionActor === p){
          Game.cancelBombAction(p);
        }
        if (canPlant){
          $('bombActionBox').style.display = 'block';
          $('bombActionText').textContent = 'Нажмите и удерживайте [E] для закладки C4 (3.0 с)';
          $('bombProgress').style.width = '0%';
        } else if (canDefuse){
          const sec = Game.getDefuseTime(p).toFixed(1);
          $('bombActionBox').style.display = 'block';
          $('bombActionText').textContent = 'Нажмите и удерживайте [E] для разминирования C4 (' + sec + ' с)';
          $('bombProgress').style.width = '0%';
        } else if (lookAtWeapon){
          $('bombActionBox').style.display = 'block';
          $('bombActionText').textContent = 'Нажмите [E] чтобы подобрать оружие';
          $('bombProgress').style.width = '0%';
        } else {
          $('bombActionBox').style.display = 'none';
        }
      }
    } else {
      if (bomb.actionActor === p) Game.cancelBombAction(p);
      $('bombActionBox').style.display = 'none';
    }

    if (Game.input.shooting && !doingBombAction && (w.rate > 200 ? !Game.input.firedOnce : true) && !st.menuOpen){
      if (st.phase === 'live') Game.playerShoot();
      Game.input.firedOnce = true;
    }
    if (!Game.input.shooting) Game.input.firedOnce = false;

    if (st.phase === 'live' && !doingBombAction){
      const isSilentWalk = keys.has('ShiftLeft');
      let baseSpd = 6.4;
      if (p.crouching) baseSpd = 2.4;
      else if (isSilentWalk) baseSpd = 3.2;

      let speed = baseSpd * (w.speedMul || 1);
      if (p.ads) speed *= p.isScoped ? 0.45 : 0.72;
      if (performance.now() < (p.slowUntil || 0)) speed *= 0.55;

      const f = new THREE.Vector3(-Math.sin(p.yaw), 0, -Math.cos(p.yaw));
      const r = new THREE.Vector3( Math.cos(p.yaw), 0, -Math.sin(p.yaw));
      const move = new THREE.Vector3();
      if (keys.has('KeyW')) move.add(f);
      if (keys.has('KeyS')) move.sub(f);
      if (keys.has('KeyD')) move.add(r);
      if (keys.has('KeyA')) move.sub(r);
      if (move.lengthSq() > 0) move.normalize().multiplyScalar(speed * dt);
      p.pos.x += move.x;
      p.pos.z += move.z;
      Game.resolveCollisions(p.pos, 0.45);

      if (keys.has('Space') && p.onGround){ p.vy = 5.4; p.onGround = false; }

      if (move.lengthSq() > 1e-6 && p.onGround && !isSilentWalk && !p.crouching){
        p.stepTimer -= dt;
        if (p.stepTimer <= 0){
          p.stepTimer = 0.34;
          Game.snd.footstep(null, 1);
        }
      } else {
        p.stepTimer = 0;
      }
    }

    const groundY = Game.getGroundHeight(p.pos.x, p.pos.z);
    p.vy -= 14 * dt;
    p.pos.y += p.vy * dt;
    if (p.pos.y <= groundY){
      p.pos.y = groundY;
      p.vy = 0;
      p.onGround = true;
    }

    Game.camera.position.set(p.pos.x, p.pos.y + p.eyeHeight, p.pos.z);
    Game.camera.rotation.y = p.yaw;
    Game.camera.rotation.x = p.pitch;
  }

    } else if (st.phase === 'live'){
      if (bomb.status === 'planted'){
        $('timer').textContent = 'C4: ' + Math.ceil(Math.max(0, bomb.timer)) + ' с';
      } else {
        st.roundTimeLeft -= dt;
        const secLeft = Math.ceil(Math.max(0, st.roundTimeLeft));
        $('timer').textContent = secLeft + ' с';

        if (secLeft <= 10 && secLeft > 0 && secLeft !== lastRoundTickSec){
          lastRoundTickSec = secLeft;
          Game.snd.roundTick();
        }

        if (st.roundTimeLeft <= 0){
          endRound(1 - st.attackTeam, 'время вышло — цель удержана');
        }
      }
    }

    hudSyncTimer -= dt;
    if (hudSyncTimer <= 0){
      hudSyncTimer = 0.2;
      Game.updHUD();
    }

    if (st.statsShown) Game.renderStats();
  }

  const clock = new THREE.Clock();
  function animate(){
    requestAnimationFrame(animate);
    const dt = Math.min(clock.getDelta(), 0.05);
    const ph = Game.state.phase;

    if (ph === 'live' || ph === 'buy'){
      updatePlayer(dt);
      if (ph === 'live'){
        for (const b of Game.bots) Game.updateBot(b, dt);
        updateBomb(dt);
      }
    } else if (ph === 'roundEnd' && Game.player.dead){
      Game.updateSpectator(dt);
    }

    updateTimers(dt);
    Game.updateGun(dt);
    Game.updateHitEffects(dt);
    Game.updateExplosions(dt);
    Game.updateGrenades(dt);
    Game.updateMinimap(dt);
    Game.updateTracers(dt);

    for (const dw of Game.droppedWeapons) dw.mesh.rotation.z += dt * 1.5;

    Game.renderer.render(Game.scene, Game.camera);
    Game.renderer.autoClear = false;
    Game.renderer.clearDepth();
    Game.renderer.render(Game.gunScene, Game.gunCam);
    Game.renderer.autoClear = true;
  }

  Game.camera.position.set(-25, 1.7, 0);
  Game.camera.rotation.y = -Math.PI / 2;
  animate();

  addEventListener('resize', () => {
    Game.camera.aspect = innerWidth / innerHeight;
    Game.camera.updateProjectionMatrix();
    Game.gunCam.aspect = innerWidth / innerHeight;
    Game.gunCam.updateProjectionMatrix();
    Game.renderer.setSize(innerWidth, innerHeight);
  });
})();
